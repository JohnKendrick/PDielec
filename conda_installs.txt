# Source-development environment (run from the PDielec repository root).
# Python 3.12 is an example choice; PDielec 10 requires Python >=3.10.
conda create --name pdielec-dev --channel conda-forge python=3.12 pip
conda activate pdielec-dev
# Install PDielec and all declared dependencies, including PyQt6 and pytest.
# pyproject.toml is the source of truth for these dependencies.
python -m pip install -e .
